import express from 'express';
import path from 'path';
import fs from 'fs';
import { fileURLToPath } from 'url';
import multer from 'multer';

const __filename = fileURLToPath(import.meta.url);
const __dirname = path.dirname(__filename);

const isProduction = process.env.NODE_ENV === 'production';

// Ensure uploads directory exists
const uploadsDir = path.resolve(__dirname, 'public/uploads');
if (!fs.existsSync(uploadsDir)) {
  fs.mkdirSync(uploadsDir, { recursive: true });
}

const videoInfoFilePath = path.join(uploadsDir, 'video-info.json');

// Configure multer storage for video uploads
const storage = multer.diskStorage({
  destination: (_req, _file, cb) => {
    cb(null, uploadsDir);
  },
  filename: (_req, file, cb) => {
    const ext = path.extname(file.originalname) || '.mp4';
    const safeName = `team_video_${Date.now()}${ext}`;
    cb(null, safeName);
  },
});

const upload = multer({
  storage,
  limits: {
    fileSize: 500 * 1024 * 1024, // 500 MB limit
  },
});

async function startServer() {
  const app = express();

  // Middleware for JSON
  app.use(express.json());

  // Serve static uploads folder explicitly
  app.use('/uploads', express.static(uploadsDir));

  // --- API Routes ---

  // 1. Get current saved video info
  app.get('/api/video-info', (_req, res) => {
    try {
      if (fs.existsSync(videoInfoFilePath)) {
        const raw = fs.readFileSync(videoInfoFilePath, 'utf-8');
        const data = JSON.parse(raw);

        // If file type, verify file still exists
        if (data.type === 'file' && data.filename) {
          const filePath = path.join(uploadsDir, data.filename);
          if (fs.existsSync(filePath)) {
            return res.json({ exists: true, video: data });
          }
        } else if (data.type === 'url' && data.videoUrl) {
          return res.json({ exists: true, video: data });
        }
      }
      return res.json({ exists: false });
    } catch (err) {
      console.error('Error reading video-info.json:', err);
      return res.json({ exists: false });
    }
  });

  // 2. Upload video file
  app.post('/api/upload-video', upload.single('video'), (req, res) => {
    try {
      if (!req.file) {
        return res.status(400).json({ success: false, error: 'No video file provided' });
      }

      // Clean up previously uploaded video files to save space
      try {
        if (fs.existsSync(videoInfoFilePath)) {
          const prev = JSON.parse(fs.readFileSync(videoInfoFilePath, 'utf-8'));
          if (prev.type === 'file' && prev.filename && prev.filename !== req.file.filename) {
            const oldPath = path.join(uploadsDir, prev.filename);
            if (fs.existsSync(oldPath)) {
              fs.unlinkSync(oldPath);
            }
          }
        }
      } catch (cleanErr) {
        console.warn('Could not clean previous video:', cleanErr);
      }

      const videoData = {
        type: 'file',
        filename: req.file.filename,
        originalName: req.file.originalname,
        fileUrl: `/uploads/${req.file.filename}`,
        size: req.file.size,
        mimeType: req.file.mimetype,
        uploadedAt: new Date().toISOString(),
      };

      fs.writeFileSync(videoInfoFilePath, JSON.stringify(videoData, null, 2), 'utf-8');

      return res.json({
        success: true,
        message: 'Video uploaded and saved to website successfully',
        video: videoData,
      });
    } catch (err) {
      console.error('Upload error:', err);
      return res.status(500).json({ success: false, error: (err as Error).message });
    }
  });

  // 3. Save cloud / YouTube / online URL
  app.post('/api/save-video-url', (req, res) => {
    try {
      const { url, title } = req.body;
      if (!url || typeof url !== 'string' || !url.trim()) {
        return res.status(400).json({ success: false, error: 'Valid URL is required' });
      }

      const videoData = {
        type: 'url',
        videoUrl: url.trim(),
        title: title?.trim() || '小組雲端影片',
        uploadedAt: new Date().toISOString(),
      };

      fs.writeFileSync(videoInfoFilePath, JSON.stringify(videoData, null, 2), 'utf-8');

      return res.json({
        success: true,
        message: 'Video URL saved successfully',
        video: videoData,
      });
    } catch (err) {
      console.error('Error saving video URL:', err);
      return res.status(500).json({ success: false, error: (err as Error).message });
    }
  });

  // 4. Delete saved video
  app.delete('/api/video', (_req, res) => {
    try {
      if (fs.existsSync(videoInfoFilePath)) {
        try {
          const data = JSON.parse(fs.readFileSync(videoInfoFilePath, 'utf-8'));
          if (data.type === 'file' && data.filename) {
            const filePath = path.join(uploadsDir, data.filename);
            if (fs.existsSync(filePath)) {
              fs.unlinkSync(filePath);
            }
          }
        } catch (_) {}
        fs.unlinkSync(videoInfoFilePath);
      }
      return res.json({ success: true });
    } catch (err) {
      console.error('Error deleting video:', err);
      return res.status(500).json({ success: false, error: (err as Error).message });
    }
  });

  // 5. Download complete source code ZIP
  app.get('/api/download-source-zip', (_req, res) => {
    const zipPath = path.resolve(__dirname, 'public/project-source.zip');
    if (fs.existsSync(zipPath)) {
      res.download(zipPath, 'stand-firm-mommy-full-code.zip');
    } else {
      res.status(404).send('ZIP file not found');
    }
  });

  // 6. Download single unified text file containing all code
  app.get('/api/download-full-code-text', (_req, res) => {
    const txtPath = path.resolve(__dirname, 'public/FULL_PROJECT_CODE.txt');
    if (fs.existsSync(txtPath)) {
      res.download(txtPath, 'FULL_PROJECT_CODE.txt');
    } else {
      res.status(404).send('Full code text file not found');
    }
  });

  // --- Vite / Static Handling ---

  if (!isProduction) {
    // Development: integrate Vite middlewares
    const { createServer: createViteServer } = await import('vite');
    const vite = await createViteServer({
      server: {
        middlewareMode: true,
        hmr: process.env.DISABLE_HMR !== 'true',
      },
      appType: 'spa',
    });
    app.use(vite.middlewares);
  } else {
    // Production: serve built static assets from dist
    const distDir = path.resolve(__dirname, 'dist');
    app.use(express.static(distDir));
    app.get('*', (_req, res) => {
      res.sendFile(path.join(distDir, 'index.html'));
    });
  }

  // Parse CLI args (e.g., --port 3000 --host 0.0.0.0 passed by npm or supervisor)
  const args = process.argv.slice(2);
  let port = process.env.PORT ? parseInt(process.env.PORT, 10) : 3000;
  let host = '0.0.0.0';

  for (let i = 0; i < args.length; i++) {
    if (args[i] === '--port' && args[i + 1]) {
      port = parseInt(args[i + 1], 10);
    }
    if (args[i] === '--host') {
      if (args[i + 1] && !args[i + 1].startsWith('-')) {
        host = args[i + 1];
      } else {
        host = '0.0.0.0';
      }
    }
  }

  app.listen(port, host, () => {
    console.log(`> Server running on http://${host}:${port} (mode: ${isProduction ? 'production' : 'development'})`);
  });
}

startServer().catch((err) => {
  console.error('Failed to start server:', err);
  process.exit(1);
});

// Web Audio API lightweight sound synthesizer (Kawaii & Arcade Game Edition)
let audioCtx: AudioContext | null = null;

function getAudioContext(): AudioContext | null {
  if (typeof window === 'undefined') return null;
  if (!audioCtx) {
    const AudioCtxClass = window.AudioContext || (window as unknown as { webkitAudioContext: typeof AudioContext }).webkitAudioContext;
    if (AudioCtxClass) {
      audioCtx = new AudioCtxClass();
    }
  }
  if (audioCtx && audioCtx.state === 'suspended') {
    audioCtx.resume().catch(() => {});
  }
  return audioCtx;
}

// Cute popping bubble sound
export function playKawaiiPop() {
  const ctx = getAudioContext();
  if (!ctx) return;
  const now = ctx.currentTime;
  const osc = ctx.createOscillator();
  const gain = ctx.createGain();

  osc.type = 'sine';
  osc.frequency.setValueAtTime(450, now);
  osc.frequency.exponentialRampToValueAtTime(900, now + 0.08);

  gain.gain.setValueAtTime(0.25, now);
  gain.gain.exponentialRampToValueAtTime(0.001, now + 0.09);

  osc.connect(gain);
  gain.connect(ctx.destination);
  osc.start(now);
  osc.stop(now + 0.1);
}

let morningAmbientOsc: OscillatorNode | null = null;
let morningAmbientGain: GainNode | null = null;

export function startMorningAmbientHum() {
  const ctx = getAudioContext();
  if (!ctx || morningAmbientOsc) return;

  const now = ctx.currentTime;
  morningAmbientOsc = ctx.createOscillator();
  morningAmbientGain = ctx.createGain();
  const filter = ctx.createBiquadFilter();

  morningAmbientOsc.type = 'sine';
  morningAmbientOsc.frequency.setValueAtTime(110, now); // Gentle low A2 hum

  filter.type = 'lowpass';
  filter.frequency.setValueAtTime(160, now);

  // Very subtle background volume (5%)
  morningAmbientGain.gain.setValueAtTime(0.001, now);
  morningAmbientGain.gain.linearRampToValueAtTime(0.04, now + 1.2);

  morningAmbientOsc.connect(filter);
  filter.connect(morningAmbientGain);
  morningAmbientGain.connect(ctx.destination);

  morningAmbientOsc.start(now);
}

export function stopMorningAmbientHum() {
  if (!morningAmbientGain || !morningAmbientOsc || !audioCtx) return;
  try {
    const now = audioCtx.currentTime;
    morningAmbientGain.gain.linearRampToValueAtTime(0.0001, now + 0.5);
    setTimeout(() => {
      if (morningAmbientOsc) {
        morningAmbientOsc.stop();
        morningAmbientOsc.disconnect();
        morningAmbientOsc = null;
      }
      morningAmbientGain = null;
    }, 600);
  } catch {
    morningAmbientOsc = null;
    morningAmbientGain = null;
  }
}

// IndexedDB persistent storage utility for large team videos

const DB_NAME = 'PregnancyTeamVideoDB';
const DB_VERSION = 1;
const STORE_NAME = 'saved_videos';
const VIDEO_KEY = 'team_presentation_video';

export interface SavedVideoRecord {
  id: string;
  blob: Blob;
  name: string;
  size: number;
  type: string;
  savedAt: number;
}

function openDB(): Promise<IDBDatabase> {
  return new Promise((resolve, reject) => {
    if (typeof window === 'undefined' || !window.indexedDB) {
      reject(new Error('IndexedDB is not supported in this environment'));
      return;
    }

    const request = indexedDB.open(DB_NAME, DB_VERSION);

    request.onupgradeneeded = (event) => {
      const db = (event.target as IDBOpenDBRequest).result;
      if (!db.objectStoreNames.contains(STORE_NAME)) {
        db.createObjectStore(STORE_NAME, { keyPath: 'id' });
      }
    };

    request.onsuccess = () => {
      resolve(request.result);
    };

    request.onerror = () => {
      reject(request.error);
    };
  });
}

export async function saveVideoToIndexedDB(
  fileOrBlob: Blob,
  metadata: { name: string; size: number; type: string }
): Promise<void> {
  const db = await openDB();
  return new Promise((resolve, reject) => {
    const tx = db.transaction(STORE_NAME, 'readwrite');
    const store = tx.objectStore(STORE_NAME);

    const record: SavedVideoRecord = {
      id: VIDEO_KEY,
      blob: fileOrBlob,
      name: metadata.name,
      size: metadata.size,
      type: metadata.type || 'video/mp4',
      savedAt: Date.now(),
    };

    const req = store.put(record);
    req.onsuccess = () => resolve();
    req.onerror = () => reject(req.error);
  });
}

export async function getVideoFromIndexedDB(): Promise<SavedVideoRecord | null> {
  try {
    const db = await openDB();
    return new Promise((resolve, reject) => {
      const tx = db.transaction(STORE_NAME, 'readonly');
      const store = tx.objectStore(STORE_NAME);
      const req = store.get(VIDEO_KEY);

      req.onsuccess = () => {
        resolve(req.result || null);
      };
      req.onerror = () => reject(req.error);
    });
  } catch (err) {
    console.warn('Failed to load video from IndexedDB:', err);
    return null;
  }
}

export async function deleteVideoFromIndexedDB(): Promise<void> {
  try {
    const db = await openDB();
    return new Promise((resolve, reject) => {
      const tx = db.transaction(STORE_NAME, 'readwrite');
      const store = tx.objectStore(STORE_NAME);
      const req = store.delete(VIDEO_KEY);
      req.onsuccess = () => resolve();
      req.onerror = () => reject(req.error);
    });
  } catch (err) {
    console.warn('Failed to delete video from IndexedDB:', err);
  }
}

export const SubwayGameView: React.FC<SubwayGameViewProps> = ({
  onSwitchToVideo
}) => {
  // Player profile with empty default name so user can name themselves
  const [profile, setProfile] = useState<PlayerProfile>({
    name: '',
    badgeStyle: 'pink_heart',
    avatar: '🤰'
  });

  const currentMomName = profile.name.trim() || '媽咪';

  const [phase, setPhase] = useState<GamePhase>('welcome');
  const [gameMode, setGameMode] = useState<GameMode>('full_tour');
  const [currentQIndex, setCurrentQIndex] = useState(0);
  const [correctCount, setCorrectCount] = useState(0);
  const [mistakeCount, setMistakeCount] = useState(0);
  const [balance, setBalance] = useState(100);
  const [tiltAngle, setTiltAngle] = useState(0);
  const [isWobbling, setIsWobbling] = useState(false);
  // Spontaneous seat offering state (讓座隨機出現)
  const [isSeated, setIsSeated] = useState(false);
  // Subway Carriage Dynamic Sway & Random Bump Engine (always naturally active)
  const [isRandomBump, setIsRandomBump] = useState(false);
  const [bumpDirection, setBumpDirection] = useState<'left' | 'right'>('left');
  const [swayDescription, setSwayDescription] = useState('∿ 捷運疾駛中 · 隨機輕微顛簸');
  const bumpTimeoutRef = useRef<NodeJS.Timeout | null>(null);

  const [selectedOption, setSelectedOption] = useState<number | null>(null);
  const [showExplanation, setShowExplanation] = useState(false);
  const [isAnswerCorrect, setIsAnswerCorrect] = useState(false);
  const [isTimeout, setIsTimeout] = useState(false);
  const [passengerBubble, setPassengerBubble] = useState<string | null>(null);
  const [broadcastText, setBroadcastText] = useState<string>('下車時請注意間隙 · Mind the gap');

  // Streak & Performance
  const [consecutiveCorrect, setConsecutiveCorrect] = useState(0);
  const [maxStreak, setMaxStreak] = useState(0);

  // Maternity Backpack Items System
  const [items, setItems] = useState<{
    cracker: number; // 50/50 排除卡
    plum: number;    // +10 秒生津酸梅
    shield: number;  // 免失衡金牌護盾
    socks: number;   // 平衡回滿彈性襪
  }>({
    cracker: 2,
    plum: 2,
    shield: 1,
    socks: 1
  });
  const [eliminatedIndices, setEliminatedIndices] = useState<number[]>([]);
  const [hasShield, setHasShield] = useState(false);

  // Interactive Mini-Games
  const [activeMiniGame, setActiveMiniGame] = useState<'balance_qte' | 'grandma_basket' | 'seat_dilemma' | null>(null);
  const [qteProgress, setQteProgress] = useState(0);
  const [qteTimeLeft, setQteTimeLeft] = useState(5);
  const qteTimerRef = useRef<NodeJS.Timeout | null>(null);

  // Seat Dilemma Random Event States
  const [dilemmaTimeLeft, setDilemmaTimeLeft] = useState(8);
  const [dilemmaOutcome, setDilemmaOutcome] = useState<SeatDilemmaOutcome | null>(null);
  const dilemmaTimerRef = useRef<NodeJS.Timeout | null>(null);

  // Broadcast Multi-Lingual Switcher
  const [broadcastLang, setBroadcastLang] = useState<'zh' | 'tw' | 'hakka' | 'en'>('zh');

  // Dynamic random subway incidents
  const [activeIncident, setActiveIncident] = useState<{
    id: string;
    title: string;
    desc: string;
    actionText?: string;
  } | null>(null);
  const [triggeredIncidents, setTriggeredIncidents] = useState<string[]>([]);

  // Fullscreen & Mobile View Modes
  const [isFullscreen, setIsFullscreen] = useState(false);
  const [mobileExpandedCarriage, setMobileExpandedCarriage] = useState(false);

  // Pure background audio/video player (ZERO UI FOOTPRINT)
  const bgAudioRef = useRef<HTMLAudioElement | null>(null);
  const bgVideoMediaRef = useRef<HTMLVideoElement | null>(null);
  const [customAudioUrl, setCustomAudioUrl] = useState<string | null>(null);
  const fileInputRef = useRef<HTMLInputElement | null>(null);

  // Countdown timer per question (15 seconds)
  const [timeLeft, setTimeLeft] = useState(QUESTION_TIME_LIMIT);
  const timerRef = useRef<NodeJS.Timeout | null>(null);

  // Target wins based on mode
  const targetWinCount =
    gameMode === 'full_tour' ? 16 :
    gameMode === 'quick_challenge' ? 8 : 999;

  // Current station
  const currentStation =
    STATIONS.find((s) => currentQIndex >= s.range[0] && currentQIndex <= s.range[1]) ||
    STATIONS[STATIONS.length - 1];

  // Fullscreen toggle
  const toggleFullscreen = () => {
    if (!document.fullscreenElement) {
      document.documentElement.requestFullscreen().then(() => setIsFullscreen(true)).catch(() => {});
    } else {
      document.exitFullscreen().then(() => setIsFullscreen(false)).catch(() => {});
    }
  };

  useEffect(() => {
    const handleFsChange = () => setIsFullscreen(!!document.fullscreenElement);
    document.addEventListener('fullscreenchange', handleFsChange);
    return () => document.removeEventListener('fullscreenchange', handleFsChange);
  }, []);

  // Keyboard shortcut listener for PC: 1~4 / A~D, Space / Enter
  useEffect(() => {
    if (phase !== 'playing') return;

    const handleKeyDown = (e: KeyboardEvent) => {
      if (e.target instanceof HTMLInputElement || e.target instanceof HTMLTextAreaElement) return;

      const key = e.key.toLowerCase();

      // QTE Spacebar support
      if (activeMiniGame === 'balance_qte') {
        if (key === ' ' || key === 'enter') {
          e.preventDefault();
          handleQteTap();
        }
        return;
      }

      // Seat Dilemma keyboard support
      if (activeMiniGame === 'seat_dilemma') {
        if (!dilemmaOutcome) {
          if (key === '1' || key === 'a') {
            e.preventDefault();
            handleSeatDilemmaChoice('self');
          } else if (key === '2' || key === 'b') {
            e.preventDefault();
            handleSeatDilemmaChoice('elder');
          } else if (key === '3' || key === 'c') {
            e.preventDefault();
            handleSeatDilemmaChoice('collaborate');
          }
        } else if (key === ' ' || key === 'enter') {
          e.preventDefault();
          handleCloseSeatDilemma();
        }
        return;
      }

      if (!showExplanation && selectedOption === null && timeLeft > 0) {
        if (key === '1' || key === 'a') handleSelectAnswer(0);
        else if (key === '2' || key === 'b') handleSelectAnswer(1);
        else if (key === '3' || key === 'c') handleSelectAnswer(2);
        else if (key === '4' || key === 'd') handleSelectAnswer(3);
      } else if (showExplanation && (key === ' ' || key === 'enter')) {
        e.preventDefault();
        if (correctCount < targetWinCount && mistakeCount < MAX_MISTAKES && currentQIndex < SUBWAY_QUESTIONS.length - 1) {
          handleNextQuestion();
        }
      }
    };

    window.addEventListener('keydown', handleKeyDown);
    return () => window.removeEventListener('keydown', handleKeyDown);
  }, [phase, showExplanation, selectedOption, timeLeft, correctCount, mistakeCount, currentQIndex, activeMiniGame, targetWinCount, dilemmaOutcome]);

  // Play audio in background
  const playBackgroundAudio = () => {
    setBroadcastText('正在背景播放關渡站到站廣播 (純音訊，無畫面)...');
    if (customAudioUrl && bgVideoMediaRef.current) {
      bgVideoMediaRef.current.currentTime = 0;
      bgVideoMediaRef.current.play().catch(() => {});
    } else if (bgAudioRef.current) {
      bgAudioRef.current.currentTime = 0;
      bgAudioRef.current.play().catch(() => {});
    }
  };

  const handleFileUpload = (e: React.ChangeEvent<HTMLInputElement>) => {
    const file = e.target.files?.[0];
    if (file) {
      const url = URL.createObjectURL(file);
      setCustomAudioUrl(url);
      setBroadcastText(`已載入你選取的影片：${file.name}（純背景播放聲音）`);
      if (bgVideoMediaRef.current) {
        bgVideoMediaRef.current.src = url;
        bgVideoMediaRef.current.currentTime = 0;
        bgVideoMediaRef.current.play().catch(() => {});
      }
    }
  };

  // Derive Stance
  let currentStance: MomStance = 'unheld';
  if (phase === 'lost' || mistakeCount >= MAX_MISTAKES) {
    currentStance = 'flying';
  } else if (phase === 'won' || isSeated) {
    currentStance = 'seated';
  } else if (correctCount >= 1) {
    currentStance = 'holding';
  } else {
    currentStance = 'unheld';
  }

  // Derive Mood
  let characterMood: CharacterMood = 'idle';
  if (phase === 'won' || currentStance === 'seated') {
    characterMood = 'seated';
  } else if (isWobbling || phase === 'lost' || currentStance === 'flying') {
    characterMood = 'wrong';
  } else if (isAnswerCorrect && showExplanation) {
    characterMood = 'correct';
  }

  // Dynamic Subway Sway & Random Realistic Bumps Engine
  // Simulates subway track curves, rail expansion joints, and acceleration forces automatically
  useEffect(() => {
    if (phase !== 'playing') return;

    // 1. Continuous natural carriage sway
    const swayInterval = setInterval(() => {
      if (isWobbling || isRandomBump) return;
      // Standing unheld experiences greater sway than holding
      const amplitude =
        currentStance === 'unheld' ? 14 :
        currentStance === 'holding' ? 4.5 : 1.5;
      const naturalSway = (Math.random() - 0.5) * amplitude;
      setTiltAngle(naturalSway);
    }, 380);

    // 2. Random intermittent track joint bumps & curve jolts (every 3.5 ~ 6 seconds)
    let bumpTimer: NodeJS.Timeout;
    const scheduleNextBump = () => {
      const delay = Math.random() * 2500 + 3500; // 3.5s ~ 6s
      bumpTimer = setTimeout(() => {
        if (phase === 'playing' && !isWobbling) {
          const dir = Math.random() > 0.5 ? 'left' : 'right';
          const bumpAngle = (dir === 'left' ? -1 : 1) * (currentStance === 'unheld' ? 16 : 6);
          setBumpDirection(dir);
          setIsRandomBump(true);
          setTiltAngle(bumpAngle);
          setSwayDescription(dir === 'left' ? '⚡ 軌道交會微晃 (偏左)' : '⚡ 軌道接縫微震 (偏右)');

          if (bumpTimeoutRef.current) clearTimeout(bumpTimeoutRef.current);
          bumpTimeoutRef.current = setTimeout(() => {
            setIsRandomBump(false);
            setSwayDescription(
              currentStance === 'unheld'
                ? '∿ 捷運疾駛中 · 站立無扶握重心不穩'
                : currentStance === 'holding'
                ? '💪 握緊立柱 · 車廂搖晃已有效緩衝'
                : '💖 安穩就座 · 平穩巡航'
            );
          }, 550);
        }
        scheduleNextBump();
      }, delay);
    };

    scheduleNextBump();

    return () => {
      clearInterval(swayInterval);
      clearTimeout(bumpTimer);
      if (bumpTimeoutRef.current) clearTimeout(bumpTimeoutRef.current);
    };
  }, [phase, isWobbling, currentStance, isRandomBump]);

  // Determine carriage shake & sway animation classes
  const carriageShakeClass = useMemo(() => {
    if (isWobbling) return 'animate-subway-wobble';
    if (isRandomBump) return 'animate-subway-bump';
    if (currentStance === 'unheld') return 'animate-subway-standing';
    if (currentStance === 'holding') return 'animate-subway-holding';
    return 'animate-subway-seated';
  }, [isWobbling, isRandomBump, currentStance]);

  // Question Countdown Timer
  useEffect(() => {
    if (phase !== 'playing' || showExplanation || activeMiniGame !== null) {
      if (timerRef.current) clearInterval(timerRef.current);
      return;
    }

    setTimeLeft(QUESTION_TIME_LIMIT);

    timerRef.current = setInterval(() => {
      setTimeLeft((prev) => {
        if (prev <= 1) {
          if (timerRef.current) clearInterval(timerRef.current);
          handleTimeExpire();
          return 0;
        }
        return prev - 1;
      });
    }, 1000);

    return () => {
      if (timerRef.current) clearInterval(timerRef.current);
    };
  }, [phase, currentQIndex, showExplanation, activeMiniGame]);

  const handleStartGame = () => {
    playBackgroundAudio();
    setPhase('playing');
    setCurrentQIndex(0);
    setCorrectCount(0);
    setMistakeCount(0);
    setBalance(100);
    setConsecutiveCorrect(0);
    setMaxStreak(0);
    setSelectedOption(null);
    setShowExplanation(false);
    setIsTimeout(false);
    setEliminatedIndices([]);
    setActiveMiniGame(null);
    setIsSeated(false);
    setHasShield(false);
    setItems({ cracker: 2, plum: 2, shield: 0, socks: 1 });
    setTimeLeft(QUESTION_TIME_LIMIT);
  };

  // Triggered when user selects an answer option
  const handleSelectAnswer = (optionIdx: number) => {
    if (selectedOption !== null || showExplanation || timeLeft <= 0) return;
    if (eliminatedIndices.includes(optionIdx)) return; // Can't choose eliminated option
    if (timerRef.current) clearInterval(timerRef.current);

    setSelectedOption(optionIdx);
    const question = SUBWAY_QUESTIONS[currentQIndex % SUBWAY_QUESTIONS.length];
    const isCorrect = optionIdx === question.correctIndex;
    setIsAnswerCorrect(isCorrect);
    setIsTimeout(false);
    setShowExplanation(true);

    if (isCorrect) {
      playQuizCorrectChime();
      setBalance(100);
      setTiltAngle(0);
      const nextCorrect = correctCount + 1;
      setCorrectCount(nextCorrect);
      const nextStreak = consecutiveCorrect + 1;
      setConsecutiveCorrect(nextStreak);
      if (nextStreak > maxStreak) setMaxStreak(nextStreak);

      if (nextStreak === 3) {
        setPassengerBubble('🔥 三連勝！全車乘客露出佩服的眼神！');
        confetti({ particleCount: 50, spread: 60, origin: { y: 0.6 } });
      }

      // Spontaneous Random Seat Offering (讓座隨機出現)
      if (!isSeated) {
        const seatOfferRoll = Math.random();
        if (seatOfferRoll < 0.32 || (nextCorrect >= 4 && seatOfferRoll < 0.55)) {
          setIsSeated(true);
          playSuccessChime();
          confetti({ particleCount: 55, spread: 65, origin: { y: 0.6 } });
          setPassengerBubble('🧍 讓座同學抬頭看見好孕胸章，立即起立讓座：「媽咪您辛苦了，請坐這裡！」💖');
          setBroadcastText('💖 乘客主動感動讓座！媽咪安心就座，享受平穩通勤！');
        } else {
          setPassengerBubble('💪 媽咪雙手穩穩握好黃色立柱！身體平衡超穩定！');
          setBroadcastText('💪 媽咪扶好立柱了！車廂前行中，留意是否有好心乘客讓座或空位！');
        }
      }

      // Spontaneous Random Shield Blessing (護盾隨機出現)
      if (!hasShield && Math.random() < 0.28) {
        setHasShield(true);
        setItems((prev) => ({ ...prev, shield: prev.shield + 1 }));
        setPassengerBubble('🛡️ 暖心隨機祝福！熱心乘客送來【好孕金牌護盾】，為媽咪隨機抵擋一次失衡！✨');
        playKawaiiCoin();
      }

      // Check Win Condition
      if (nextCorrect >= targetWinCount || currentQIndex >= SUBWAY_QUESTIONS.length - 1) {
        setTimeout(() => {
          handleTriggerWin();
        }, 1300);
      }
    } else {
      playQuizWrongChime();
      setConsecutiveCorrect(0);
      handleRecordMistake(false);
    }
  };

  // Time expired handler
  const handleTimeExpire = () => {
    setIsTimeout(true);
    setIsAnswerCorrect(false);
    setShowExplanation(true);
    setConsecutiveCorrect(0);
    playQuizWrongChime();
    handleRecordMistake(true);
  };

  // Handle mistake accumulation
  const handleRecordMistake = (byTimeout: boolean) => {
    if (hasShield) {
      setHasShield(false);
      setPassengerBubble('🛡️ 好孕金牌胸章發威！成功抵擋一次急煞失衡！');
      setBroadcastText('🛡️ 媽咪緊握立柱，護盾抵擋了晃動！化險為夷！');
      setIsWobbling(false);
      playSuccessChime();
      return;
    }

    setIsWobbling(true);
    setTiltAngle(Math.random() > 0.5 ? -35 : 35);

    const nextMistakes = mistakeCount + 1;
    setMistakeCount(nextMistakes);
    setBalance(nextMistakes >= MAX_MISTAKES ? 0 : nextMistakes === 3 ? 25 : nextMistakes === 2 ? 50 : 75);

    setTimeout(() => {
      setIsWobbling(false);
      setTiltAngle(0);
    }, 1000);

    if (nextMistakes >= MAX_MISTAKES) {
      setPassengerBubble('😱 全車乘客：「天哪！列車一個大急煞～媽咪飛出去了！」');
      setBroadcastText('🚨 警報！累積失衡 4 次急煞！媽咪直接重心不穩飛出車廂！');
      setTimeout(() => {
        setPhase('lost');
      }, 1400);
    }
  };

  const handleNextQuestion = () => {
    playKawaiiPop();
    setSelectedOption(null);
    setShowExplanation(false);
    setIsTimeout(false);
    setEliminatedIndices([]);
    const nextQ = currentQIndex + 1;
    setCurrentQIndex(nextQ);

    // Station Transition Check
    if (nextQ === 5 || nextQ === 10 || nextQ === 15 || nextQ === 20) {
      playSubwayStationChime();
      playMetroDoorBeep();
      // Trigger Balance QTE Mini-Game at station curve!
      triggerBalanceQte();
      return;
    }

    // Seat Dilemma Random Event Check: occurs randomly when not seated
    if (!isSeated && !triggeredIncidents.includes('seat_dilemma') && (nextQ === 3 || (nextQ >= 2 && Math.random() < 0.35))) {
      setTriggeredIncidents((prev) => [...prev, 'seat_dilemma']);
      triggerSeatDilemma();
      return;
    }

    // Dynamic incidents at other stations
    if (nextQ === 2 && !triggeredIncidents.includes('curve')) {
      setTriggeredIncidents((prev) => [...prev, 'curve']);
      setActiveIncident({
        id: 'curve',
        title: '⚠️ 突發狀況：列車即將大過彎！',
        desc: '前方大彎道！車廂離心力晃動，請立即握穩黃色立柱！',
        actionText: '💪 抓緊立柱！'
      });
    } else if (nextQ === 7 && !triggeredIncidents.includes('plum')) {
      setTriggeredIncidents((prev) => [...prev, 'plum']);
      setActiveIncident({
        id: 'plum',
        title: '👵 暖心小巧思：陳阿嬤遞來生津話梅！',
        desc: '陳阿嬤看媽咪有點孕吐臉色微白：「妹妹吃顆話梅含著生津止渴！」失衡容錯次數 +1，答題時間延長 +5 秒！',
        actionText: '💖 謝謝陳阿嬤！'
      });
    } else if (nextQ === 12 && !triggeredIncidents.includes('dog')) {
      setTriggeredIncidents((prev) => [...prev, 'dog']);
      setActiveIncident({
        id: 'dog',
        title: '🐶 萌寵突發：推車裡的小柴犬豆豆探頭！',
        desc: '推車裡的小柴犬探出毛茸茸小臉蛋汪了一聲，吐出小舌頭，全車乘客被融化！緊張氣氛消除，答題時間延長 +5 秒！',
        actionText: '🥰 摸摸小柴犬！'
      });
    } else if (nextQ === 17 && !triggeredIncidents.includes('broadcast')) {
      setTriggeredIncidents((prev) => [...prev, 'broadcast']);
      setHasShield(true);
      setItems((prev) => ({ ...prev, shield: prev.shield + 1 }));
      setActiveIncident({
        id: 'broadcast',
        title: '📢 暖心廣播：北捷站務廣播同理心宣導！',
        desc: '廣播響起：「車廂內請主動關心身邊需要座位的旅客，感謝您的愛心。」全車乘客好感度暴增，隨機獲贈【好孕護盾 x1】！',
        actionText: '✨ 感受車廂溫暖！'
      });
    } else if (nextQ === 22 && !triggeredIncidents.includes('kick')) {
      setTriggeredIncidents((prev) => [...prev, 'kick']);
      setActiveIncident({
        id: 'kick',
        title: '👶 溫馨小彩蛋：肚子裡的寶寶踢了一腳！',
        desc: '『媽咪站得好穩，好棒！』肚子冒出滿滿愛心光芒，平衡值加滿，精神抖擻！',
        actionText: '🥰 摸摸肚肚繼續加油！'
      });
    }
  };

  // Balance QTE Mini-Game
  const triggerBalanceQte = () => {
    setActiveMiniGame('balance_qte');
    setQteProgress(20);
    setQteTimeLeft(5);

    if (qteTimerRef.current) clearInterval(qteTimerRef.current);
    qteTimerRef.current = setInterval(() => {
      setQteTimeLeft((prev) => {
        if (prev <= 1) {
          if (qteTimerRef.current) clearInterval(qteTimerRef.current);
          handleFinishQte(false);
          return 0;
        }
        return prev - 1;
      });
    }, 1000);
  };

  const handleQteTap = () => {
    playKawaiiPop();
    setQteProgress((prev) => {
      const next = prev + 18;
      if (next >= 100) {
        if (qteTimerRef.current) clearInterval(qteTimerRef.current);
        handleFinishQte(true);
        return 100;
      }
      return next;
    });
  };

  const handleFinishQte = (success: boolean) => {
    if (success) {
      playSuccessChime();
      confetti({ particleCount: 80, spread: 70, origin: { y: 0.6 } });
      setBalance(100);
      setItems((prev) => ({ ...prev, shield: prev.shield + 1 }));
      setPassengerBubble('🎉 平衡挑戰大成功！四平八穩無傷通過，獲得【好孕護盾 x1】！');
      setBroadcastText('💪 媽咪身手矯健，獲得全車乘客一致起立鼓掌！');
    } else {
      playKawaiiOuch();
      setPassengerBubble('😵 列車太晃啦！稍微踉蹌了一下，但媽咪還是努力站穩了！');
    }
    setTimeout(() => {
      setActiveMiniGame(null);
    }, 1100);
  };

  // Seat Dilemma Event Trigger
  const triggerSeatDilemma = () => {
    playMetroDoorBeep();
    playKawaiiPop();
    setActiveMiniGame('seat_dilemma');
    setDilemmaOutcome(null);
    setDilemmaTimeLeft(8);

    if (dilemmaTimerRef.current) clearInterval(dilemmaTimerRef.current);
    dilemmaTimerRef.current = setInterval(() => {
      setDilemmaTimeLeft((prev) => {
        if (prev <= 1) {
          if (dilemmaTimerRef.current) clearInterval(dilemmaTimerRef.current);
          handleSeatDilemmaChoice('timeout');
          return 0;
        }
        return prev - 1;
      });
    }, 1000);
  };

  const handleSeatDilemmaChoice = (choice: 'self' | 'elder' | 'collaborate' | 'timeout') => {
    if (dilemmaTimerRef.current) clearInterval(dilemmaTimerRef.current);

    if (choice === 'self') {
      playSuccessChime();
      confetti({ particleCount: 70, spread: 60, origin: { y: 0.6 } });
      setIsSeated(true);
      setBalance(100);
      setHasShield(true);
      setItems((prev) => ({ ...prev, shield: prev.shield + 1 }));

function formatFileSize(bytes?: number): string {
  if (!bytes) return '';
  const k = 1024;
  const sizes = ['B', 'KB', 'MB', 'GB'];
  const i = Math.floor(Math.log(bytes) / Math.log(k));
  return parseFloat((bytes / Math.pow(k, i)).toFixed(1)) + ' ' + sizes[i];
}

function getYouTubeEmbedUrl(url: string): string | null {
  const match = url.match(
    /(?:youtu\.be\/|youtube\.com\/(?:embed\/|v\/|watch\?v=|watch\?.+&v=))([\w-]{11})/
  );
  if (match && match[1]) {
    return `https://www.youtube.com/embed/${match[1]}?autoplay=1&rel=0`;
  }
  return null;
}

function getGoogleDriveEmbedUrl(url: string): string | null {
  const match = url.match(/drive\.google\.com\/file\/d\/([a-zA-Z0-9_-]+)/);
  if (match && match[1]) {
    return `https://drive.google.com/file/d/${match[1]}/preview`;
  }
  return null;
}

        console.warn('IndexedDB check error:', idbErr);
      }

      if (isMounted) {
        setIsLoading(false);
      }
    }

    loadSavedVideo();

    return () => {
      isMounted = false;
      if (blobUrlRef.current) {
        URL.revokeObjectURL(blobUrlRef.current);
      }
    };
  }, []);

  // Handle local file upload (stores in both IndexedDB and server)
  const handleFileUpload = async (e: React.ChangeEvent<HTMLInputElement>) => {
    const file = e.target.files?.[0];
    if (!file) return;

    playKawaiiPop();
    setUploadError('');
    setUploadProgress(10);
    setStatusMessage('準備儲存影片中...');

    // 1. Immediately create local blob URL so user sees instant preview
    if (blobUrlRef.current) {
      URL.revokeObjectURL(blobUrlRef.current);
    }
    const localBlobUrl = URL.createObjectURL(file);
    blobUrlRef.current = localBlobUrl;

    const newVideoState: SavedVideoState = {
      type: 'file',
      url: localBlobUrl,
      name: file.name,
      size: file.size,
      uploadedAt: new Date().toISOString(),
      savedInIndexedDB: false,
      savedOnServer: false,
    };
    setVideoData(newVideoState);

    // 2. Save directly to client IndexedDB for permanent local browser storage
    try {
      setStatusMessage('正在儲存至本機資料庫 (IndexedDB)...');
      await saveVideoToIndexedDB(file, {
        name: file.name,
        size: file.size,
        type: file.type || 'video/mp4',
      });
      newVideoState.savedInIndexedDB = true;
      setVideoData({ ...newVideoState });
    } catch (err) {
      console.warn('IndexedDB save failed:', err);
    }

    // 3. Upload to server with progress tracking so anyone visiting the link can view it
    try {
      setStatusMessage('正在上傳並保存在網站伺服器...');
      setUploadProgress(30);

      const formData = new FormData();
      formData.append('video', file);

      const xhr = new XMLHttpRequest();

      xhr.upload.onprogress = (event) => {
        if (event.lengthComputable) {
          const percent = Math.round((event.loaded / event.total) * 60) + 30;
          setUploadProgress(Math.min(percent, 95));
        }
      };

      xhr.onload = () => {
        if (xhr.status >= 200 && xhr.status < 300) {
          try {
            const resp = JSON.parse(xhr.responseText);
            if (resp.success && resp.video) {
              setUploadProgress(100);
              setStatusMessage('影片已成功保存在本網站！下次點開此連結將自動播放！');
              setVideoData({
                type: 'file',
                url: resp.video.fileUrl,
                name: resp.video.originalName,
                size: resp.video.size,
                uploadedAt: resp.video.uploadedAt,
                savedOnServer: true,
                savedInIndexedDB: true,
              });
              setTimeout(() => {
                setUploadProgress(null);
                setStatusMessage('');
              }, 4000);
              return;
            }
          } catch (_) {}
        }
        // Fallback: If server rejects because file is too big or disk limit,
        // it's already safely preserved in IndexedDB!
        setUploadProgress(null);
        setStatusMessage('已保存在本瀏覽器中！下次開啟此連結仍可繼續播放。');
        setTimeout(() => setStatusMessage(''), 4000);
      };

      xhr.onerror = () => {
        setUploadProgress(null);
        setStatusMessage('已保存在本瀏覽器中！下次開啟此連結仍可繼續播放。');
        setTimeout(() => setStatusMessage(''), 4000);
      };

      xhr.open('POST', '/api/upload-video');
      xhr.send(formData);
    } catch (uploadErr) {
      console.warn('Server upload error:', uploadErr);
      setUploadProgress(null);
      setStatusMessage('已保存在本瀏覽器資料庫中！');
      setTimeout(() => setStatusMessage(''), 4000);
    }

    // Reset input so re-selecting same file triggers onChange
    if (e.target) {
      e.target.value = '';
    }
  };

  // Handle saving cloud URL (YouTube, Google Drive, MP4 URL)
  const handleSaveUrl = async () => {
    if (!inputUrl.trim()) return;

    playKawaiiPop();
    setIsLoading(true);
    setUploadError('');

    try {
      const res = await fetch('/api/save-video-url', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          url: inputUrl.trim(),
          title: inputTitle.trim() || '小組雲端影片',
        }),
      });

      if (res.ok) {
        const json = await res.json();
        if (json.success && json.video) {
          setVideoData({
            type: 'url',
            url: json.video.videoUrl,
            name: json.video.title,
            uploadedAt: json.video.uploadedAt,
            savedOnServer: true,
          });
          setShowUrlModal(false);
          setInputUrl('');
          setInputTitle('');
          setStatusMessage('✅ 線上影片連結已儲存！所有點開此網站連結的人都能觀看！');
          setTimeout(() => setStatusMessage(''), 4000);
          setIsLoading(false);
          return;
        }
      }
    } catch (err) {
      console.warn('Failed to save URL to server:', err);
    }

    // Fallback: save to state
    setVideoData({
      type: 'url',
      url: inputUrl.trim(),
      name: inputTitle.trim() || '小組雲端影片',
      uploadedAt: new Date().toISOString(),
      savedOnServer: false,
    });
    setShowUrlModal(false);
    setIsLoading(false);
  };

  // Clear / Delete saved video
  const handleClearVideo = async () => {
    if (!window.confirm('確定要移除此小組影片嗎？')) return;

    playKawaiiPop();
    setIsLoading(true);

    try {
      // 1. Delete from server
      await fetch('/api/video', { method: 'DELETE' });
    } catch (err) {
      console.warn('Server delete error:', err);
    }

    try {
      // 2. Delete from IndexedDB
      await deleteVideoFromIndexedDB();
    } catch (err) {
      console.warn('IndexedDB delete error:', err);
    }

    if (blobUrlRef.current) {
      URL.revokeObjectURL(blobUrlRef.current);
      blobUrlRef.current = null;
    }

    setVideoData(null);
    setIsLoading(false);
    setStatusMessage('影片已自網站與本機儲存空間清除。');
    setTimeout(() => setStatusMessage(''), 3000);
  };

  const handleCopyLink = () => {
    playKawaiiPop();
    navigator.clipboard.writeText(window.location.href);
    setCopiedLink(true);
    setTimeout(() => setCopiedLink(false), 2500);
  };

  // Render video or embedded player
  const renderVideoPlayer = () => {
    if (!videoData) return null;

    if (videoData.type === 'url') {
      const ytEmbed = getYouTubeEmbedUrl(videoData.url);
      if (ytEmbed) {
        return (
          <iframe
            src={ytEmbed}
            title={videoData.name}
            className="w-full h-full border-0"
            allow="accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture"
            allowFullScreen
          />
        );
      }

      const driveEmbed = getGoogleDriveEmbedUrl(videoData.url);
      if (driveEmbed) {
        return (
          <iframe
            src={driveEmbed}
            title={videoData.name}
            className="w-full h-full border-0"
            allow="autoplay"
            allowFullScreen
          />
        );
      }

      // Direct URL fallback
      return (
        <video
          src={videoData.url}
          controls
          autoPlay
          className="w-full h-full object-contain"
        />
      );
    }

    // Direct file
    return (
      <video
        src={videoData.url}
        controls
        autoPlay
        playsInline
        className="w-full h-full object-contain"
      />
    );
  };

  return (
    <div className="w-full max-w-[1700px] mx-auto px-3 sm:px-6 lg:px-8 py-4 sm:py-6 space-y-5">
      {/* Hidden file input */}
      <input
        ref={fileInputRef}
        type="file"
        accept="video/*"
        onChange={handleFileUpload}
        className="hidden"
      />

      {/* Header */}
      <div className="bg-gradient-to-r from-[#D3E8F4] via-[#FFE0E6] to-[#FFFCF4] rounded-3xl border-2 border-[#FFBCBD] p-5 sm:p-8 shadow-xs flex flex-col sm:flex-row items-center justify-between gap-4">
        <div>
          <div className="inline-flex items-center gap-1.5 bg-[#E85D75] text-white text-xs font-black px-3.5 py-1 rounded-full shadow-xs mb-1.5">
            <Film className="w-3.5 h-3.5" />
            <span>成果發表專屬 · 小組實測成果展演</span>
          </div>
          <h1 className="text-2xl sm:text-3xl font-black text-[#4A3337] font-serif-tc">
            小組妊娠體驗實測短片
          </h1>
          <p className="text-xs sm:text-sm text-[#6B4B52] mt-1">
            上傳或連結你們小組拍攝的真實影片檔，<strong>永久保存在網站中</strong>，下次點開連結隨時播放！
          </p>
        </div>

        <div className="flex items-center gap-2">
          <button
            onClick={handleCopyLink}
            className="px-3.5 py-2.5 bg-white text-[#4A3337] hover:bg-[#FFE0E6] rounded-2xl text-xs font-black border border-[#FFBCBD] shadow-xs cursor-pointer shrink-0 flex items-center gap-1.5 transition-all"
            title="複製目前網址，分享給老師或同學觀看"
          >
            {copiedLink ? (
              <>
                <Check className="w-3.5 h-3.5 text-emerald-600" />
                <span className="text-emerald-700">網址已複製！</span>
              </>
            ) : (
              <>
                <Copy className="w-3.5 h-3.5 text-[#E85D75]" />
                <span>複製分享網址</span>
              </>
            )}
          </button>

          <button
            onClick={onBackToGame}
            className="px-4 py-2.5 bg-[#FFFCF4] text-[#4A3337] rounded-2xl text-xs font-black border border-[#FFBCBD] hover:bg-[#FFE0E6] shadow-xs cursor-pointer shrink-0 transition-colors"
          >
            ← 返回捷運答題遊戲
          </button>
        </div>
      </div>

      {/* Status banner if message exists */}
      {statusMessage && (
        <div className="bg-emerald-50 border-2 border-emerald-300 text-emerald-800 p-3.5 rounded-2xl flex items-center gap-2.5 text-xs font-bold animate-in fade-in shadow-xs">
          <CheckCircle2 className="w-4 h-4 text-emerald-600 shrink-0" />
          <span>{statusMessage}</span>
        </div>
      )}

      {/* Upload progress indicator */}
      {uploadProgress !== null && (
        <div className="bg-[#FFFCF4] border-2 border-[#FFBCBD] p-4 rounded-2xl space-y-2 shadow-xs">
          <div className="flex justify-between items-center text-xs font-black text-[#4A3337]">
            <span className="flex items-center gap-1.5">
              <Loader2 className="w-3.5 h-3.5 text-[#E85D75] animate-spin" />
              正在保存影片至網站...
            </span>
            <span className="font-mono text-[#E85D75]">{uploadProgress}%</span>
          </div>
          <div className="w-full bg-stone-200 h-2.5 rounded-full overflow-hidden">
            <div
              className="bg-gradient-to-r from-[#E85D75] to-[#FF8C94] h-full transition-all duration-300"
              style={{ width: `${uploadProgress}%` }}
            />
          </div>
          <p className="text-[11px] text-[#6B4B52]">
            影片正同步寫入網站伺服器與本機快取，保存完成後下次點開網址即可自動載入！
          </p>
        </div>
      )}

      {/* Upload & Control Action Bar */}
      <div className="bg-[#FFFCF4] rounded-3xl border-2 border-[#FFBCBD] p-4 shadow-xs flex flex-wrap items-center justify-between gap-3 text-xs">
        <div className="flex items-center gap-2.5 flex-wrap">
          <div className="p-2 rounded-xl bg-[#FFE0E6] text-[#E85D75]">
            <FileVideo className="w-4 h-4" />
          </div>

          {isLoading ? (
            <span className="font-bold text-[#6B4B52] flex items-center gap-1.5">
              <Loader2 className="w-3.5 h-3.5 animate-spin" />
              正在確認網站已保存的影片...
            </span>
          ) : videoData ? (
            <div className="space-y-0.5">
              <div className="flex items-center gap-2 flex-wrap">
                <span className="font-bold text-[#4A3337]">
                  目前保存影片：<span className="text-[#E85D75] font-mono">{videoData.name}</span>
                </span>
                {videoData.size && (
                  <span className="text-[10px] bg-stone-100 text-stone-600 px-2 py-0.5 rounded-md font-mono">
                    {formatFileSize(videoData.size)}
                  </span>
                )}
              </div>
              <div className="flex items-center gap-2 text-[10px] text-emerald-700 font-bold">
                <CheckCircle2 className="w-3 h-3 text-emerald-600 inline" />
                <span>
                  已保存在本網站
                  {videoData.savedOnServer ? '（伺服器同步存檔）' : ''}
                  {videoData.savedInIndexedDB ? '（本機快取備份）' : ''}，下次開啟此連結自動載入！
                </span>
              </div>
            </div>
          ) : (
            <div className="space-y-0.5">
              <span className="font-bold text-[#6B4B52]">
                尚未儲存小組影片，請選擇影片檔案或輸入線上影片網址：
              </span>
              <p className="text-[10px] text-stone-500">
                支援 MP4、MOV、WebM 或 YouTube / Google Drive，保存後下次點開連結直接觀看！
              </p>
            </div>
          )}
        </div>

        {/* Action Buttons */}
        <div className="flex items-center gap-2 flex-wrap">
          {/* File Upload Button */}
          <button
            onClick={() => fileInputRef.current?.click()}
            className="px-4 py-2 bg-[#E85D75] hover:bg-[#D44763] active:scale-95 text-white rounded-xl font-black cursor-pointer flex items-center gap-1.5 transition-all shadow-xs"
          >
            <Upload className="w-3.5 h-3.5" />
            <span>{videoData ? '更換影片檔案' : '上傳小組影片檔案'}</span>
          </button>

          {/* Cloud / YouTube URL Button */}
          <button
            onClick={() => {
              playKawaiiPop();
              setShowUrlModal(!showUrlModal);
            }}
            className="px-3.5 py-2 bg-[#D3E8F4] hover:bg-[#BCDDF0] text-[#30556E] active:scale-95 rounded-xl font-black cursor-pointer flex items-center gap-1.5 transition-all border border-[#BCDDF0]"
          >
            <Link2 className="w-3.5 h-3.5" />
            <span>{videoData?.type === 'url' ? '更換線上網址' : '輸入線上/雲端連結'}</span>
          </button>

          {/* Download Button (for file) */}
          {videoData && videoData.type === 'file' && (
            <a
              href={videoData.url}
              download={videoData.name}
              className="px-3 py-2 bg-stone-100 hover:bg-stone-200 text-[#4A3337] rounded-xl font-bold cursor-pointer flex items-center gap-1 transition-all border border-stone-200 text-xs"
              title="下載影片備份檔"
            >
              <Download className="w-3.5 h-3.5 text-stone-600" />
              <span>下載</span>
            </a>
          )}

          {/* Delete Button */}
          {videoData && (
            <button
              onClick={handleClearVideo}
              className="px-3 py-2 bg-[#FFE0E6] hover:bg-[#FFBCBD] text-rose-800 rounded-xl font-bold cursor-pointer flex items-center gap-1 transition-all border border-[#FFBCBD]"
              title="刪除已保存的影片"
            >
              <Trash2 className="w-3.5 h-3.5" />
              <span>移除影片</span>
            </button>
          )}
        </div>
      </div>

      {/* Online / Cloud Link Input Modal Panel */}
      {showUrlModal && (
        <div className="bg-[#FFFCF4] rounded-3xl border-2 border-[#BCDDF0] p-5 shadow-md animate-in fade-in space-y-3">
          <div className="flex items-center justify-between">
            <div className="flex items-center gap-2">
              <Link2 className="w-4 h-4 text-[#30556E]" />
              <h3 className="font-black text-[#30556E] text-sm">
                儲存線上影片網址（YouTube / Google 雲端硬碟 / 影片直連）
              </h3>
            </div>
            <button
              onClick={() => setShowUrlModal(false)}
              className="text-stone-400 hover:text-stone-600 text-xs font-bold"
            >
              ✕ 關閉
            </button>
          </div>

          <p className="text-xs text-[#6B4B52]">
            如果影片檔案較大（例如超過 300MB），可以直接將影片上傳至 YouTube（可設為不公開）或 Google 雲端硬碟（開啟連結共用），將網址貼在下方，同樣會保存在網站中，下次開啟連結就能直接播放！
          </p>

          <div className="grid grid-cols-1 sm:grid-cols-2 gap-3 pt-1">
            <div>
              <label className="block text-[11px] font-black text-[#4A3337] mb-1">
                影片網址 (URL) *
              </label>
              <input
                type="url"
                value={inputUrl}
                onChange={(e) => setInputUrl(e.target.value)}
                placeholder="https://youtu.be/... 或 https://drive.google.com/..."
                className="w-full px-3 py-2 text-xs border border-stone-300 rounded-xl focus:outline-none focus:border-[#E85D75] bg-white font-mono"
              />
            </div>
            <div>
              <label className="block text-[11px] font-black text-[#4A3337] mb-1">
                小組影片標題 (選填)
              </label>
              <input
                type="text"
                value={inputTitle}
                onChange={(e) => setInputTitle(e.target.value)}
                placeholder="例：第一組 台北捷運孕婦負重體驗實測短片"
                className="w-full px-3 py-2 text-xs border border-stone-300 rounded-xl focus:outline-none focus:border-[#E85D75] bg-white"
              />
            </div>
          </div>

          <div className="flex justify-end gap-2 pt-2">
            <button
              onClick={() => setShowUrlModal(false)}
              className="px-3.5 py-1.5 text-xs text-stone-600 hover:bg-stone-100 rounded-xl font-bold"
            >
              取消
            </button>
            <button
              onClick={handleSaveUrl}
              disabled={!inputUrl.trim()}
              className="px-4 py-1.5 text-xs bg-[#30556E] hover:bg-[#233F53] disabled:opacity-50 text-white rounded-xl font-black flex items-center gap-1.5 shadow-xs cursor-pointer"
            >
              <Check className="w-3.5 h-3.5" />
              <span>保存線上影片至網站</span>
            </button>
          </div>
        </div>
      )}

      {/* Video Player Display Screen */}
      <div className="bg-stone-950 rounded-3xl border-3 border-[#FFBCBD] overflow-hidden shadow-xl aspect-16/9 relative flex items-center justify-center">
        {isLoading ? (
          <div className="flex flex-col items-center justify-center gap-3 text-stone-300">
            <Loader2 className="w-10 h-10 text-[#FFBCBD] animate-spin" />
            <p className="text-sm font-bold">載入已保存影片中...</p>
          </div>
        ) : videoData ? (
          renderVideoPlayer()
        ) : (
          /* Empty State Dropzone / Selector */
          <div
            onClick={() => fileInputRef.current?.click()}
            className="w-full h-full flex flex-col items-center justify-center p-8 text-center cursor-pointer hover:bg-stone-900 transition-colors group"
          >
            <div className="w-20 h-20 rounded-3xl bg-[#D3E8F4]/20 border-2 border-[#D3E8F4] group-hover:scale-105 transition-transform flex items-center justify-center text-4xl mb-4 shadow-lg">
              <Upload className="w-8 h-8 text-[#D3E8F4]" />
            </div>
            <h3 className="text-xl font-black font-serif-tc text-[#FFE0E6] mb-1">
              點擊此處上傳你們小組的實測影片檔案
            </h3>
            <p className="text-xs text-stone-400 max-w-md mb-4">
              支援手機攝影錄影檔、.mp4、.mov、.webm 等格式。上傳後自動保存在網站，下次點開連結就能直接看！
            </p>
            <div className="flex items-center gap-2.5">
              <span className="px-5 py-2.5 bg-[#E85D75] group-hover:bg-[#D44763] text-white rounded-full font-bold text-xs shadow-md flex items-center gap-1.5">
                <Upload className="w-3.5 h-3.5" />
                <span>📂 選擇本地影片並永久保存</span>
              </span>
              <button
                type="button"
                onClick={(e) => {
                  e.stopPropagation();
                  setShowUrlModal(true);
                }}
                className="px-4 py-2.5 bg-stone-800 hover:bg-stone-700 text-[#D3E8F4] rounded-full font-bold text-xs border border-stone-600 flex items-center gap-1.5"
              >
                <Link2 className="w-3.5 h-3.5" />
                <span>貼上線上連結</span>
              </button>
            </div>
          </div>
        )}
      </div>

    </div>
  );
};
