import express from 'express';
import http from 'http';
import path from 'path';
import fs from 'fs';
import { fileURLToPath } from 'url';
import multer from 'multer';

const __filename = fileURLToPath(import.meta.url);
const __dirname = path.dirname(__filename);

const isProduction = process.env.NODE_ENV === 'production';

// Ensure uploads directory exists
const uploadsDir = path.resolve(__dirname, 'public/uploads');
if (!fs.existsSync(uploadsDir)) {
  fs.mkdirSync(uploadsDir, { recursive: true });
}

const videoInfoFilePath = path.join(uploadsDir, 'video-info.json');

// Configure multer storage for video uploads
const storage = multer.diskStorage({
  destination: (_req, _file, cb) => {
    cb(null, uploadsDir);
  },
  filename: (_req, file, cb) => {
    const ext = path.extname(file.originalname) || '.mp4';
    const safeName = `team_video_${Date.now()}${ext}`;
    cb(null, safeName);
  },
});

const upload = multer({
  storage,
  limits: {
    fileSize: 500 * 1024 * 1024, // 500 MB limit
  },
});

async function startServer() {
  const app = express();
  const server = http.createServer(app);

  // Middleware for JSON
  app.use(express.json());

  // Serve static uploads folder explicitly
  app.use('/uploads', express.static(uploadsDir));

  const persistentJsonPath = path.resolve(__dirname, 'src/data/persistedVideo.json');

  function isOldNews(d: any): boolean {
    if (!d) return false;
    const s = JSON.stringify(d);
    return s.includes('_HHZWmEm9K4') || s.includes('TVBS') || s.includes('新聞') || s.includes('挺肚搭捷運');
  }

  function getPersistedVideo() {
    // 1. First check uploadsDir/video-info.json
    try {
      if (fs.existsSync(videoInfoFilePath)) {
        const raw = fs.readFileSync(videoInfoFilePath, 'utf-8');
        const data = JSON.parse(raw);
        if (isOldNews(data)) {
          fs.unlinkSync(videoInfoFilePath);
        } else if (data && data.type === 'file' && data.filename) {
          const filePath = path.join(uploadsDir, data.filename);
          if (fs.existsSync(filePath)) {
            return data;
          }
        } else if (data && data.type === 'url' && data.videoUrl) {
          return data;
        }
      }
    } catch (e) {
      console.warn('Error reading videoInfoFilePath:', e);
    }

    // 2. Second check persistentJsonPath (committed in repo)
    try {
      if (fs.existsSync(persistentJsonPath)) {
        const raw = fs.readFileSync(persistentJsonPath, 'utf-8');
        const data = JSON.parse(raw);
        if (isOldNews(data)) {
          return null;
        } else if (data && data.type === 'url' && (data.videoUrl || data.url)) {
          return {
            ...data,
            videoUrl: data.videoUrl || data.url,
            title: data.title || data.name || '小組實測影片',
          };
        } else if (data && data.type === 'file' && data.fileUrl) {
          return data;
        }
      }
    } catch (e) {
      console.warn('Error reading persistentJsonPath:', e);
    }

    return null;
  }

  function savePersistedVideo(videoData: any) {
    try {
      fs.writeFileSync(videoInfoFilePath, JSON.stringify(videoData, null, 2), 'utf-8');
    } catch (e) {
      console.warn('Could not write to videoInfoFilePath:', e);
    }
    try {
      fs.writeFileSync(persistentJsonPath, JSON.stringify(videoData, null, 2), 'utf-8');
    } catch (e) {
      console.warn('Could not write to persistentJsonPath:', e);
    }
  }

  // --- API Routes ---

  // 1. Get current saved video info
  app.get('/api/video-info', (_req, res) => {
    try {
      const video = getPersistedVideo();
      if (video) {
        return res.json({ exists: true, video });
      }
      return res.json({ exists: false });
    } catch (err) {
      console.error('Error in /api/video-info:', err);
      return res.json({ exists: false });
    }
  });

  // 2. Upload video file
  app.post('/api/upload-video', upload.single('video'), (req, res) => {
    try {
      if (!req.file) {
        return res.status(400).json({ success: false, error: 'No video file provided' });
      }

      // Clean up previously uploaded video files to save space
      try {
        if (fs.existsSync(videoInfoFilePath)) {
          const prev = JSON.parse(fs.readFileSync(videoInfoFilePath, 'utf-8'));
          if (prev.type === 'file' && prev.filename && prev.filename !== req.file.filename) {
            const oldPath = path.join(uploadsDir, prev.filename);
            if (fs.existsSync(oldPath)) {
              fs.unlinkSync(oldPath);
            }
          }
        }
      } catch (cleanErr) {
        console.warn('Could not clean previous video:', cleanErr);
      }

      const videoData = {
        type: 'file',
        filename: req.file.filename,
        originalName: req.file.originalname,
        title: req.file.originalname,
        fileUrl: `/uploads/${req.file.filename}`,
        size: req.file.size,
        mimeType: req.file.mimetype,
        uploadedAt: new Date().toISOString(),
        isDefault: false,
      };

      savePersistedVideo(videoData);

      return res.json({
        success: true,
        message: '影片檔案已成功保存在網站伺服器！所有訪客都能同步看到此影片！',
        video: videoData,
      });
    } catch (err) {
      console.error('Upload error:', err);
      return res.status(500).json({ success: false, error: (err as Error).message });
    }
  });

  // 3. Save cloud / YouTube / online URL
  app.post('/api/save-video-url', (req, res) => {
    try {
      const { url, title, description } = req.body;
      if (!url || typeof url !== 'string' || !url.trim()) {
        return res.status(400).json({ success: false, error: 'Valid URL is required' });
      }

      const videoData = {
        type: 'url',
        videoUrl: url.trim(),
        title: title?.trim() || '小組實測影片',
        description: description?.trim() || '',
        uploadedAt: new Date().toISOString(),
        isDefault: false,
      };

      savePersistedVideo(videoData);

      return res.json({
        success: true,
        message: '線上影片網址已成功永久保存在網站中！所有點開連結的人都能同步觀看！',
        video: videoData,
      });
    } catch (err) {
      console.error('Error saving video URL:', err);
      return res.status(500).json({ success: false, error: (err as Error).message });
    }
  });

  // 4. Delete saved video
  app.delete('/api/video', (_req, res) => {
    try {
      if (fs.existsSync(videoInfoFilePath)) {
        try {
          const data = JSON.parse(fs.readFileSync(videoInfoFilePath, 'utf-8'));
          if (data && data.type === 'file' && data.filename) {
            const filePath = path.join(uploadsDir, data.filename);
            if (fs.existsSync(filePath)) {
              fs.unlinkSync(filePath);
            }
          }
        } catch (_) {}
        try {
          fs.unlinkSync(videoInfoFilePath);
        } catch (_) {}
      }

      try {
        fs.writeFileSync(persistentJsonPath, JSON.stringify({ type: null }, null, 2), 'utf-8');
      } catch (_) {}

      return res.json({
        success: true,
        message: '影片已移除',
      });
    } catch (err) {
      console.error('Error deleting video:', err);
      return res.status(500).json({ success: false, error: (err as Error).message });
    }
  });

  // 5. Download complete source code ZIP
  app.get('/api/download-source-zip', (_req, res) => {
    const zipPath = path.resolve(__dirname, 'public/project-source.zip');
    if (fs.existsSync(zipPath)) {
      res.download(zipPath, 'stand-firm-mommy-full-code.zip');
    } else {
      res.status(404).send('ZIP file not found');
    }
  });

  // 6. Download single unified text file containing all code
  app.get('/api/download-full-code-text', (_req, res) => {
    const txtPath = path.resolve(__dirname, 'public/FULL_PROJECT_CODE.txt');
    if (fs.existsSync(txtPath)) {
      res.download(txtPath, 'FULL_PROJECT_CODE.txt');
    } else {
      res.status(404).send('Full code text file not found');
    }
  });

  // --- Vite / Static Handling ---

  if (!isProduction) {
    // Development: integrate Vite middlewares
    const { createServer: createViteServer } = await import('vite');
    const vite = await createViteServer({
      server: {
        middlewareMode: true,
        hmr: false,
      },
      appType: 'spa',
    });
    app.use(vite.middlewares);
  } else {
    // Production: serve built static assets from dist
    const distDir = path.resolve(__dirname, 'dist');
    app.use(express.static(distDir));
    app.get('*', (_req, res) => {
      res.sendFile(path.join(distDir, 'index.html'));
    });
  }

  // Parse CLI args (e.g., --port 3000 --host 0.0.0.0 passed by npm or supervisor)
  const args = process.argv.slice(2);
  let port = process.env.PORT ? parseInt(process.env.PORT, 10) : 3000;
  let host = '0.0.0.0';

  for (let i = 0; i < args.length; i++) {
    if (args[i] === '--port' && args[i + 1]) {
      port = parseInt(args[i + 1], 10);
    }
    if (args[i] === '--host') {
      if (args[i + 1] && !args[i + 1].startsWith('-')) {
        host = args[i + 1];
      } else {
        host = '0.0.0.0';
      }
    }
  }

  server.listen(port, host, () => {
    console.log(`> Server running on http://${host}:${port} (mode: ${isProduction ? 'production' : 'development'})`);
  });
}

startServer().catch((err) => {
  console.error('Failed to start server:', err);
  process.exit(1);
});

// Web Audio API lightweight sound synthesizer (Kawaii & Arcade Game Edition)
let audioCtx: AudioContext | null = null;

function getAudioContext(): AudioContext | null {
  if (typeof window === 'undefined') return null;
  if (!audioCtx) {
    const AudioCtxClass = window.AudioContext || (window as unknown as { webkitAudioContext: typeof AudioContext }).webkitAudioContext;
    if (AudioCtxClass) {
      audioCtx = new AudioCtxClass();
    }
  }
  if (audioCtx && audioCtx.state === 'suspended') {
    audioCtx.resume().catch(() => {});
  }
  return audioCtx;
}

// Cute popping bubble sound
export function playKawaiiPop() {
  const ctx = getAudioContext();
  if (!ctx) return;
  const now = ctx.currentTime;
  const osc = ctx.createOscillator();
  const gain = ctx.createGain();

  osc.type = 'sine';
  osc.frequency.setValueAtTime(450, now);
  osc.frequency.exponentialRampToValueAtTime(900, now + 0.08);

  gain.gain.setValueAtTime(0.25, now);
  gain.gain.exponentialRampToValueAtTime(0.001, now + 0.09);

  osc.connect(gain);
  gain.connect(ctx.destination);
  osc.start(now);
  osc.stop(now + 0.1);
}

// Gentle Morning Ambient Hum (輕柔清晨臥室環境微音)
let morningAmbientOsc: OscillatorNode | null = null;
let morningAmbientGain: GainNode | null = null;

export function startMorningAmbientHum() {
  const ctx = getAudioContext();
  if (!ctx || morningAmbientOsc) return;

  const now = ctx.currentTime;
  morningAmbientOsc = ctx.createOscillator();
  morningAmbientGain = ctx.createGain();
  const filter = ctx.createBiquadFilter();

  morningAmbientOsc.type = 'sine';
  morningAmbientOsc.frequency.setValueAtTime(110, now); // Gentle low A2 hum

  filter.type = 'lowpass';
  filter.frequency.setValueAtTime(160, now);

  // Very subtle background volume (5%)
  morningAmbientGain.gain.setValueAtTime(0.001, now);
  morningAmbientGain.gain.linearRampToValueAtTime(0.04, now + 1.2);

  morningAmbientOsc.connect(filter);
  filter.connect(morningAmbientGain);
  morningAmbientGain.connect(ctx.destination);

  morningAmbientOsc.start(now);
}

export function stopMorningAmbientHum() {
  if (!morningAmbientGain || !morningAmbientOsc || !audioCtx) return;
  try {
    const now = audioCtx.currentTime;
    morningAmbientGain.gain.linearRampToValueAtTime(0.0001, now + 0.5);
    setTimeout(() => {
      if (morningAmbientOsc) {
        morningAmbientOsc.stop();
        morningAmbientOsc.disconnect();
        morningAmbientOsc = null;
      }
      morningAmbientGain = null;
    }, 600);
  } catch {
    morningAmbientOsc = null;
    morningAmbientGain = null;
  }
}

// IndexedDB & LocalStorage persistent storage utility for large team videos

const DB_NAME = 'PregnancyTeamVideoDB';
const DB_VERSION = 2;
const STORE_NAME = 'saved_videos';
const VIDEO_KEY = 'team_presentation_video';
const LOCAL_STORAGE_KEY = 'team_persisted_video_meta';

export interface SavedVideoRecord {
  id: string;
  blob?: Blob;
  buffer?: ArrayBuffer;
  name: string;
  size: number;
  type: string;
  savedAt: number;
}

function openDB(): Promise<IDBDatabase> {
  return new Promise((resolve, reject) => {
    if (typeof window === 'undefined' || !window.indexedDB) {
      reject(new Error('IndexedDB is not supported in this environment'));
      return;
    }

    const request = indexedDB.open(DB_NAME, DB_VERSION);

    request.onupgradeneeded = (event) => {
      const db = (event.target as IDBOpenDBRequest).result;
      if (!db.objectStoreNames.contains(STORE_NAME)) {
        db.createObjectStore(STORE_NAME, { keyPath: 'id' });
      }
    };

    request.onsuccess = () => {
      resolve(request.result);
    };

    request.onerror = () => {
      reject(request.error);
    };
  });
}

export async function saveVideoToIndexedDB(
  fileOrBlob: Blob,
  metadata: { name: string; size: number; type: string }
): Promise<void> {
  try {
    const db = await openDB();
    
    // Safely convert to ArrayBuffer to prevent Safari/iOS DataCloneError with Blobs
    let arrayBuffer: ArrayBuffer | null = null;
    try {
      arrayBuffer = await fileOrBlob.arrayBuffer();
    } catch (_) {
      // Fallback if arrayBuffer() is unavailable
    }

    return new Promise((resolve, reject) => {
      const tx = db.transaction(STORE_NAME, 'readwrite');
      const store = tx.objectStore(STORE_NAME);

      const record: SavedVideoRecord = {
        id: VIDEO_KEY,
        buffer: arrayBuffer || undefined,
        blob: arrayBuffer ? undefined : fileOrBlob,
        name: metadata.name,
        size: metadata.size,
        type: metadata.type || 'video/mp4',
        savedAt: Date.now(),
      };

      const req = store.put(record);
      req.onsuccess = () => resolve();
      req.onerror = () => reject(req.error);
    });
  } catch (err) {
    console.warn('saveVideoToIndexedDB error:', err);
  }
}

export async function getVideoFromIndexedDB(): Promise<{ blob: Blob; name: string; size: number; type: string; savedAt: number } | null> {
  try {
    const db = await openDB();
    return new Promise((resolve, reject) => {
      const tx = db.transaction(STORE_NAME, 'readonly');
      const store = tx.objectStore(STORE_NAME);
      const req = store.get(VIDEO_KEY);

      req.onsuccess = () => {
        const record = req.result as SavedVideoRecord | undefined;
        if (!record) {
          resolve(null);
          return;
        }

        let blob: Blob | null = null;
        if (record.buffer) {
          blob = new Blob([record.buffer], { type: record.type || 'video/mp4' });
        } else if (record.blob) {
          blob = record.blob;
        }

        if (blob) {
          resolve({
            blob,
            name: record.name,
            size: record.size,
            type: record.type,
            savedAt: record.savedAt,
          });
        } else {
          resolve(null);
        }
      };
      req.onerror = () => reject(req.error);
    });
  } catch (err) {
    console.warn('Failed to load video from IndexedDB:', err);
    return null;
  }
}

          const bumpAngle = (dir === 'left' ? -1 : 1) * (currentStance === 'unheld' ? 16 : 6);

          }, 550);
        }
        scheduleNextBump();
      }, delay);
    };

    scheduleNextBump();

    return () => {
      clearInterval(swayInterval);
      clearTimeout(bumpTimer);
      if (bumpTimeoutRef.current) clearTimeout(bumpTimeoutRef.current);

                  {dilemmaOutcome.rewardText}
                </div>

                {/* Return Button */}
                <button
                  type="button"
                  onClick={handleCloseSeatDilemma}
                  className="w-full py-3.5 bg-[#E85D75] hover:bg-[#D44763] active:scale-98 text-white rounded-2xl text-sm sm:text-base font-black shadow-md cursor-pointer transition-all flex items-center justify-center gap-2"
                >
                  <Heart className="w-5 h-5 fill-white" />
                  <span>💖 帶著感動繼續通勤（完成抉擇）</span>

      )}

          WIN SCREEN WITH COMPREHENSIVE COMMUTE REPORT CARD
         ========================================================================= */}
      {phase === 'won' && (
        <div className="bg-gradient-to-b from-[#D3E8F4]/80 via-[#FFE0E6] to-[#FFFCF4] rounded-3xl border-3 border-[#FFBCBD] p-6 sm:p-10 shadow-xl text-center space-y-5 max-w-3xl mx-auto animate-in zoom-in-95 duration-200">
          <div className="flex justify-center">
            <PregnantMomIllustration
              mood="seated"
              name={currentMomName}
              badgeStyle={profile.badgeStyle}
              tiltAngle={0}
              stance="seated"
            />
          </div>

          <div className="space-y-2">
            <span className="inline-block bg-[#E85D75] text-white text-xs font-black px-4 py-1.5 rounded-full shadow-xs">
              COMMUTE MASTER · 溫馨讓座大成功！
            </span>
            <h2 className="text-xl sm:text-3xl font-black text-[#4A3337] font-serif-tc">
              🎉 抵達目的地！乘客感動讓座，{currentMomName}安心入座！
            </h2>
            <p className="text-xs sm:text-sm text-[#6B4B52] max-w-lg mx-auto leading-relaxed font-medium">
              經歷了忠義站、關渡站、竹圍站、紅樹林站到淡水終點站，{currentMomName}在顛簸車廂中雙手抓穩立柱，中途受到買菜阿嬤的及時相助，並在累積答對 7 題後感動全車起立讓座，順利攻破所有產科關卡！
            </p>
          </div>

          <div className="bg-[#FFFCF4] rounded-2xl border-2 border-[#FFBCBD] p-4 sm:p-5 text-left max-w-lg mx-auto space-y-3 shadow-xs">
            <div className="text-xs font-black text-[#4A3337] border-b border-[#FFBCBD]/40 pb-2 flex items-center justify-between">
              <span>📋 捷運通勤護理成就報告單</span>
              <span className="text-[#E85D75] font-mono">滿分守護！</span>
            </div>
            <div className="grid grid-cols-3 gap-2 text-center text-xs">
              <div className="p-2 bg-[#FFE0E6]/40 rounded-xl">
                <span className="text-[10px] text-[#7A5860] block font-bold">答對題數</span>

  type: 'file' | 'url';

// Check if a video entry is any residual news video so we can purge it
function isOldNewsVideo(v: any): boolean {
  if (!v) return false;
  const str = JSON.stringify(v);
  return (
    str.includes('_HHZWmEm9K4') ||
    str.includes('TVBS') ||
    str.includes('新聞') ||
    str.includes('news') ||
    str.includes('台北捷運孕婦負重體驗') ||
    str.includes('實測！ 孕婦挺肚搭捷運')
  );
}

export function getYouTubeEmbedUrl(url: string): string | null {
  if (!url) return null;
  const shortsMatch = url.match(/youtube\.com\/shorts\/([a-zA-Z0-9_-]{11})/);
  if (shortsMatch && shortsMatch[1]) {
    return `https://www.youtube.com/embed/${shortsMatch[1]}?autoplay=0&rel=0`;
  }
  const match = url.match(
    /(?:youtu\.be\/|youtube\.com\/(?:embed\/|v\/|watch\?v=|watch\?.+&v=))([a-zA-Z0-9_-]{11})/
  );
  if (match && match[1]) {
    return `https://www.youtube.com/embed/${match[1]}?autoplay=0&rel=0`;
  }
  return null;
}

export function getGoogleDriveEmbedUrl(url: string): string | null {
  if (!url) return null;
  const match = url.match(/drive\.google\.com\/file\/d\/([a-zA-Z0-9_-]+)/);
  if (match && match[1]) {
    return `https://drive.google.com/file/d/${match[1]}/preview`;
  }
  return null;
}

              <Link2 className="w-3.5 h-3.5 text-[#30556E]" />
              <h3 className="font-bold text-[#30556E] text-xs">
                貼上線上影片網址（YouTube / Google 雲端硬碟 / 直連）
              </h3>
            </div>
            <button
              onClick={() => setShowUrlModal(false)}
              className="text-stone-400 hover:text-stone-600 text-xs font-bold px-1"
            >
              ✕
            </button>
          </div>

          <div className="space-y-2">
            <div>
              <input
                type="url"
                value={inputUrl}
                onChange={(e) => setInputUrl(e.target.value)}
                placeholder="貼上網址：https://youtu.be/... 或 雲端連結"
                className="w-full px-3 py-1.5 text-xs border border-stone-300 rounded-lg focus:outline-none focus:border-[#E85D75] bg-white font-mono"
              />
            </div>
            <div>
              <input
                type="text"
                value={inputTitle}
                onChange={(e) => setInputTitle(e.target.value)}
                placeholder="影片名稱 (選填，如：第一組實測短片)"
                className="w-full px-3 py-1.5 text-xs border border-stone-300 rounded-lg focus:outline-none focus:border-[#E85D75] bg-white"
              />
            </div>
          </div>

          <div className="flex justify-end gap-2 pt-0.5">
            <button
              onClick={() => setShowUrlModal(false)}
              className="px-3 py-1 text-xs text-stone-600 hover:bg-stone-100 rounded-lg"
            >
              取消
            </button>
            <button
              onClick={handleSaveUrl}
              disabled={!inputUrl.trim()}
              className="px-3 py-1 text-xs bg-[#30556E] hover:bg-[#233F53] disabled:opacity-50 text-white rounded-lg font-bold cursor-pointer"
            >
              確認儲存
            </button>
          </div>
        </div>
      )}

      {/* Video Player Display Screen */}
      <div className="bg-stone-950 rounded-2xl border-2 border-[#FFBCBD] overflow-hidden shadow-md aspect-video w-full relative flex items-center justify-center">
        {isLoading ? (
          <div className="flex flex-col items-center justify-center gap-2 text-stone-300">
            <Loader2 className="w-6 h-6 text-[#FFBCBD] animate-spin" />
            <p className="text-xs font-bold">載入中...</p>
          </div>
        ) : videoData ? (
          renderVideoPlayer()
        ) : (
          /* Empty State: Clean upload prompt */
          <div
            onClick={() => fileInputRef.current?.click()}
            className="w-full h-full flex flex-col items-center justify-center p-4 text-center cursor-pointer hover:bg-stone-900 transition-colors group"
          >
            <div className="w-12 h-12 sm:w-14 sm:h-14 rounded-2xl bg-[#FFE0E6]/20 border border-[#FFBCBD] group-hover:scale-105 transition-transform flex items-center justify-center mb-2.5 shadow-sm">
              <Upload className="w-6 h-6 text-[#FFBCBD]" />
            </div>
            <h3 className="text-sm sm:text-base font-black text-[#FFE0E6] mb-1">
              點擊上傳小組實測影片
            </h3>
            <p className="text-[11px] text-stone-400 max-w-xs mb-3">
              支援手機錄影檔 (.mp4, .mov, .webm) 或線上連結
            </p>
            <div className="flex items-center gap-2">
              <span className="px-3 py-1.5 bg-[#E85D75] group-hover:bg-[#D44763] text-white rounded-xl font-bold text-xs shadow-xs flex items-center gap-1">
                <Upload className="w-3 h-3" />
                <span>選擇影片檔案</span>
              </span>
              <button
                type="button"
                onClick={(e) => {
                  e.stopPropagation();
                  setShowUrlModal(true);
                }}
                className="px-3 py-1.5 bg-stone-800 hover:bg-stone-700 text-stone-300 rounded-xl font-bold text-xs border border-stone-600 flex items-center gap-1"
              >
                <Link2 className="w-3 h-3" />
                <span>貼上網址</span>
              </button>
            </div>
          </div>
        )}
      </div>

      {/* Simple Action Bar under Video */}
      <div className="flex items-center justify-between flex-wrap gap-2 text-xs pt-0.5">
        <div className="flex items-center gap-2">
          <button
            onClick={() => fileInputRef.current?.click()}
            className="px-3 py-1.5 bg-[#E85D75] hover:bg-[#D44763] text-white rounded-lg font-bold cursor-pointer flex items-center gap-1 shadow-xs transition-colors"
          >
            <Upload className="w-3.5 h-3.5" />
            <span>{videoData ? '更換影片' : '上傳影片檔案'}</span>
          </button>

          <button
            onClick={() => {
              playKawaiiPop();
              setShowUrlModal(!showUrlModal);
            }}
            className="px-3 py-1.5 bg-[#FFFCF4] hover:bg-[#FFE0E6] text-[#4A3337] rounded-lg font-bold cursor-pointer flex items-center gap-1 border border-[#FFBCBD] transition-colors"
          >
            <Link2 className="w-3.5 h-3.5 text-[#E85D75]" />
            <span>貼上線上網址</span>
          </button>
        </div>

        {videoData && (
          <div className="flex items-center gap-1.5">
            {videoData.type === 'file' && (
              <a
                href={videoData.url}
                download={videoData.name}
                className="px-2 py-1 text-stone-600 hover:text-stone-800 hover:bg-stone-100 rounded-lg font-medium flex items-center gap-1"
                title="下載影片"
              >
                <Download className="w-3 h-3" />
                <span>下載</span>
              </a>
            )}
            <button
              onClick={handleClearVideo}
              className="px-2 py-1 text-stone-500 hover:text-rose-700 hover:bg-rose-50 rounded-lg font-medium flex items-center gap-1 transition-colors"
            >
              <Trash2 className="w-3 h-3" />
              <span>移除</span>
            </button>
          </div>
        )}
      </div>
    </div>
  );
};
